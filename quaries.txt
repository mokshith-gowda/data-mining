-- Create database tables for master data
CREATE TABLE stores (
    store_id VARCHAR(10) PRIMARY KEY,
    store_name VARCHAR(100),
    region VARCHAR(50),
    address TEXT
);

CREATE TABLE product_categories (
    category_id VARCHAR(10) PRIMARY KEY,
    category_name VARCHAR(100)
);

CREATE TABLE products (
    product_code VARCHAR(20) PRIMARY KEY,
    product_name VARCHAR(100),
    category_id VARCHAR(10) REFERENCES product_categories(category_id)
);

-- Table for point-in-time historical pricing (Task D)
CREATE TABLE price_revisions (
    revision_id SERIAL PRIMARY KEY,
    product_code VARCHAR(20) REFERENCES products(product_code),
    effective_price DECIMAL(10,2),
    valid_from TIMESTAMP,
    valid_to TIMESTAMP
);

# 2. Federated Query with As-Of Temporal Pricing Join (Tasks D & E)
query = """
    SELECT 
        p.category_name,
        SUM(sales.quantity * pr.effective_price) AS total_revenue
    FROM v_sales sales
    JOIN pg.products pr_master ON sales.product_code = pr_master.product_code
    JOIN pg.product_categories p ON pr_master.category_id = p.category_id
    -- Point-in-time price join using price revisions table (Task D)
    JOIN pg.price_revisions pr ON sales.product_code = pr.product_code
     AND sales.timestamp >= pr.valid_from 
     AND (sales.timestamp < pr.valid_to OR pr.valid_to IS NULL)
    WHERE sales.line_type = 'SALE'
      AND sales.year = '2024' 
      AND sales.month = '10'
    GROUP BY 1
    ORDER BY total_revenue DESC;
"""
